Задание 2
x = float(input("Введите x: "))
y = float(input("Введите y: "))
if (x**2 + y**2 <= 1) and (y <= x):
    print("Точка принадлежит заштрихованной области")
else:
    print("Точка не принадлежит заштрихованной области")

Задание 3
kx = int(input("Введите координату x короля: "))
ky = int(input("Введите координату y короля: "))
threat = [] # список угрожающих ладей
for i in range(1, 4):
    rx = int(input(f"Введите координату x ладьи {i}: "))
    ry = int(input(f"Введите координату y ладьи {i}: "))
    if rx == kx or ry == ky:
        threat.append(i)
if threat:
    print("Король под боем! Угрожают ладьи:", threat)
else:
    print("Король не под боем")
